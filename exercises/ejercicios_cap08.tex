% ── E1: Producción de OH y estados de oxidación ────────────────────────────
\item La producci\'on de OH depende de la cadena
\ref{RA}--\ref{RC}.

\begin{enumerate}[label=\alph*)]
    \item Indique el papel de cada reacci\'on: cu\'al aporta
          \ce{O(^1D)}, cu\'al lo desactiva por colisi\'on
          (reacci\'on \ref{RB}) y cu\'al genera OH
          (reacci\'on \ref{RC}).

    \item En la troposfera baja, $k_2[\mathrm{M}]
          \approx 10^8$~s$^{-1}$ y
          $k_3[\ce{H2O}] \approx 10^5$~s$^{-1}$.
          Calcule la fracci\'on de \ce{O(^1D)} que finalmente
          produce OH en lugar de desactivarse.

    \item Si $[\ce{O3}] = 40$~ppb $= 1.0\times10^{12}$
          mol\'ec/cm$^3$ y la constante fotol\'itica es
          $k_1 = 3\times10^{-5}$~s$^{-1}$, calcule la tasa
          de producci\'on de \ce{O(^1D)} y,
          usando la fracci\'on del inciso (b),
          la tasa neta de producci\'on de OH.

    \item El cap\'itulo indica que el estado de oxidaci\'on
          de todos los elementos en su estado libre es 0.
          Calcule el estado de oxidaci\'on del S en
          \ce{SO2}, \ce{SO3} y \ce{H2SO4}, y del N en
          NO, \ce{NO2} y \ce{HNO3}.
          ?`En qu\'e direcci\'on cambia el estado de
          oxidaci\'on al pasar de \ce{SO2} a \ce{H2SO4}?
\end{enumerate}

\textit{[Respuesta:
(b) $f = 10^5/(10^8+10^5) \approx 0.1\%$;
(c) $P_{\ce{O(^1D)}} = 3\times10^{-5}\times10^{12}
    = 3\times10^7$~mol\'ec~cm$^{-3}$~s$^{-1}$;
    $P_\mathrm{OH} = 2\times0.001\times3\times10^7
    = 6\times10^4$~mol\'ec~cm$^{-3}$~s$^{-1}$;
(d) S en \ce{SO2}: $+4$; \ce{SO3}: $+6$; \ce{H2SO4}: $+6$;
    N en NO: $+2$; \ce{NO2}: $+4$; \ce{HNO3}: $+5$;
    el S se oxida de $+4$ a $+6$]}

% ── E2: Cadena de oxidación — clasificación de etapas ──────────────────────
\item La cadena de oxidaci\'on se divide en iniciaci\'on,
propagaci\'on, ramificaci\'on y terminaci\'on
(reacciones \ref{R1}--\ref{R2} del cap\'itulo).

\begin{enumerate}[label=\alph*)]
    \item Clasifique cada reacci\'on seg\'un la etapa
          que representa:
          \begin{enumerate}[label=\roman*)]
              \item \ce{HCHO + h{\nu} -> H^. + CHO^.}
              \item \ce{OH^. + C4H10 -> H2O + C4H9^.}
              \item \ce{HO2^. + HO2^. -> H2O2 + O2}
              \item \ce{H2O2 + h{\nu} -> 2OH^.}
          \end{enumerate}

    \item Los COV act\'uan como ``combustible'' en la
          propagaci\'on. Con el \textbf{Cuadro~\ref{tvcov}},
          indique cu\'al de los siguientes se consume m\'as
          r\'apidamente con OH y cu\'al persiste m\'as tiempo
          en la atm\'osfera: metano, propeno, isopreno.

    \item Explique por qu\'e la etapa de terminaci\'on
          requiere colisi\'on entre dos radicales y por qu\'e
          esto hace que sea poco frecuente respecto
          a la propagaci\'on.

    \item El cap\'itulo indica que los principales oxidantes
          atmosf\'ericos son OH, \ce{O3} y \ce{NO3}
          en orden decreciente de fuerza oxidante.
          Para cada uno, indique si opera principalmente
          de d\'ia, de noche o en ambos momentos.
\end{enumerate}

\textit{[Respuesta:
(a-i) iniciaci\'on; (a-ii) propagaci\'on;
      (a-iii) terminaci\'on; (a-iv) ramificaci\'on;
(b) m\'as r\'apido: isopreno (2.8~h);
    m\'as persistente: metano (1837~d);
(d) OH: d\'ia; \ce{O3}: d\'ia y noche; \ce{NO3}: noche]}

% ── E3: Vida química de COVs — cálculo numérico y escala espacial ───────────
\item Con los datos del \textbf{Cuadro~\ref{tvcov}}
y $[\ce{OH}] = 1\times10^6$~mol\'ec/cm$^3$:

\begin{enumerate}[label=\alph*)]
    \item Calcule la constante de velocidad $k_{\ce{OH},i}$
          (cm$^3$~mol\'ec$^{-1}$~s$^{-1}$) para propeno
          ($\tau=10.6$~h) e isopreno ($\tau=2.8$~h),
          usando $k_i = 1/(\tau\,[\ce{OH}])$.

    \item La escala de influencia espacial de un COV es
          $L = u\cdot\tau$ con $u = \SI{10}{\metre\per\second}$.
          Calcule $L$ para metano, butano y $\beta$-pineno.
          Clasifique cada uno como local ($L<100$~km),
          regional (100--1000~km) o global ($L>1000$~km).

    \item El eteno tiene $\tau_{\ce{OH}} = 1.4$~d y
          $\tau_{\ce{O3}} = 6.7$~d.
          ?`Cu\'al oxidante domina su destrucci\'on durante
          el d\'ia? ?`Y de noche, cuando \ce{OH}
          pr\'acticamente desaparece?

    \item El $\beta$-pineno reacciona con \ce{NO3} en 12~min.
          Calcule la distancia m\'axima que recorre
          de noche con $u = \SI{5}{\metre\per\second}$
          antes de ser destruido.
\end{enumerate}

\textit{[Respuesta:
(a) propeno: $k = 2.62\times10^{-11}$;
    isopreno: $k = 9.92\times10^{-11}$~cm$^3$~mol\'ec$^{-1}$~s$^{-1}$;
(b) metano: $\approx 1.6\times10^6$~km (global);
    butano: $\approx 4\,150$~km (global/regional);
    $\beta$-pineno: $\approx 126$~km (regional);
(d) $L = 5\times720 = 3\,600$~m $= 3.6$~km (local)]}

% ── E4: Química del SO₂ — rutas de oxidación ──────────────────────────────
\item El \ce{SO2} se oxida principalmente por la ruta
del radical OH en fase gas.

\begin{enumerate}[label=\alph*)]
    \item Escriba la secuencia completa:
          \ce{SO2 + OH^. + M ->},
          \ce{HOSO2^. + O2 ->},
          \ce{SO3 + H2O ->},
          identificando el agente oxidante inicial,
          el radical que regenera \ce{HO_$x$}
          y el producto \'acido final.

    \item La reacci\'on directa \ce{2SO2 + O2 -> 2SO3}
          tiene $K_{eq}\approx8\times10^{11}$ a \SI{25}{\celsius}
          pero es cin\'eticamente muy lenta en fase gas.
          Explique con el concepto de energ\'ia de activaci\'on
          por qu\'e una $K_{eq}$ alta no garantiza una
          reacci\'on r\'apida.

    \item El tiempo de vida del \ce{SO2} frente a OH es
          $\tau\approx7$~d con
          $k = 9\times10^{-13}$~cm$^3$~mol\'ec$^{-1}$~s$^{-1}$.
          Calcule $[\ce{OH}]$ medio que produce ese tiempo
          de vida y compare con la concentraci\'on
          t\'ipica global ($1.2\times10^6$~mol\'ec/cm$^3$).

    \item Compare la vida del \ce{H2S} ($\approx70$~h)
          con la del \ce{SO2} ($\approx7$~d).
          ?`Cu\'al produce \ce{H2SO4} m\'as r\'apidamente
          cerca de una fuente industrial?
          ?`Cu\'al viaja m\'as lejos antes de oxidarse?
\end{enumerate}

\textit{[Respuesta:
(c) $[\ce{OH}] = 1/(7\times86\,400\times9\times10^{-13})
    = 1.82\times10^6$~mol\'ec/cm$^3$;
    cercano al valor global de $1.2\times10^6$;
(d) \ce{H2S} produce \ce{SO2}/\ce{H2SO4} m\'as r\'apido;
    \ce{SO2} viaja m\'as lejos ($\tau$~7 veces mayor)]}

% ── E5: Estado fotoestacionario NO₂/NO/O₃ — cálculo completo ──────────────
\item El estado fotoestacionario del sistema
\ce{NO2}/NO/\ce{O3} (reacciones \ref{RN1}--\ref{RN3})
lleva a la relaci\'on~\ref{RN8}:
$[\ce{O3}]_\mathrm{ss} = (k_1/k_3)\,([\ce{NO2}]/[\ce{NO}])$.

\begin{enumerate}[label=\alph*)]
    \item Con $k_1/k_3 = 10$~ppb y
          $[\ce{NO2}]/[\ce{NO}] = 5$ (aire urbano al mediod\'ia),
          calcule $[\ce{O3}]_\mathrm{ss}$.

    \item Usando la f\'ormula \ref{RN10} con
          $[\ce{NO2}]_0 = 100$~ppb y
          $[\ce{O3}]_0 = [\ce{NO}]_0 = 0$,
          calcule $[\ce{O3}]$ y verifique que coincide con
          el valor del \textbf{Cuadro~\ref{ozono_foto}}.

    \item Repita el c\'alculo del inciso (b) con
          $[\ce{NO2}]_0 = 1000$~ppb y compare con la
          segunda fila del \textbf{Cuadro~\ref{ozono_foto}}.

    \item En aire urbano real, $[\ce{O3}]$ observado
          (120~ppb) supera ampliamente al calculado.
          Nombre dos reacciones o procesos adicionales
          que no figuran en \ref{RN1}--\ref{RN3}
          y que producen \ce{O3} en exceso.
          ?`Qu\'e papel juegan los COV en este exceso?
\end{enumerate}

\textit{[Respuesta:
(a) $[\ce{O3}]_\mathrm{ss} = 10\times5 = 50$~ppb;
(b) $[\ce{O3}] = \tfrac{1}{2}(\sqrt{100+4000}-10)
    \approx 27$~ppb~\checkmark;
(c) $[\ce{O3}] = \tfrac{1}{2}(\sqrt{100+40\,000}-10)
    \approx 95$~ppb~\checkmark]}

% ── E6: DMS y compuestos reducidos de azufre ───────────────────────────────
\item El sulfuro de dimetilo (DMS, \ce{CH3SCH3}) es la
principal fuente natural de azufre al oc\'eano global.

\begin{enumerate}[label=\alph*)]
    \item A bajas latitudes (d\'ia):
          $k_{\ce{DMS+OH}} = 5\times10^{-12}$~cm$^3$~mol\'ec$^{-1}$~s$^{-1}$
          y $[\ce{OH}] = 3\times10^6$~mol\'ec/cm$^3$.
          Calcule $\tau_\mathrm{OH}$ del DMS.

    \item A altas latitudes (noche invernal):
          $k_{\ce{DMS+NO3}} = 1\times10^{-12}$~cm$^3$~mol\'ec$^{-1}$~s$^{-1}$
          y $[\ce{NO3}] = 5\times10^8$~mol\'ec/cm$^3$.
          Calcule $\tau_\mathrm{NO3}$ del DMS.

    \item Compare ambos $\tau$ e indique en qu\'e
          regi\'on/\'epoca el DMS persiste m\'as tiempo.
          ?`C\'omo afecta esa mayor persistencia a la
          formaci\'on de aerosoles de sulfato y,
          por consiguiente, a la reflexi\'on de luz solar
          sobre el oc\'eano?

    \item El cap\'itulo indica que el \ce{H2S} tiene
          $\tau\approx70$~h frente a OH.
          Calcule $k_{\ce{H2S+OH}}$ con
          $[\ce{OH}] = 1\times10^6$~mol\'ec/cm$^3$
          y compare con $k_{\ce{DMS+OH}}$.
\end{enumerate}

\textit{[Respuesta:
(a) $\tau_\mathrm{OH} = 1/(5\times10^{-12}
    \times3\times10^6) \approx 6.7\times10^4$~s
    $\approx$ 18.5~h;
(b) $\tau_\mathrm{NO3} = 1/(10^{-12}
    \times5\times10^8) = 2\times10^3$~s $\approx$ 33~min;
(c) mayor persistencia a altas latitudes (d\'ia invernal);
(d) $k_{\ce{H2S+OH}} = 1/(70\times3600\times10^6)
    \approx 3.97\times10^{-12}$~cm$^3$~mol\'ec$^{-1}$~s$^{-1}$;
    similar al del DMS]}

% ── E7: Aminas, PAN y compuestos de carbono ────────────────────────────────
\item Las aminas y los compuestos org\'anicos de carbono
tienen rutas de oxidaci\'on caracter\'isticas.

\begin{enumerate}[label=\alph*)]
    \item El \ce{NH3} es muy soluble y reacciona con el
          \'acido n\'itrico gaseoso para formar
          \ce{NH4NO3_{(s)}}. Si se emiten
          \SI{100}{\kilo\gram} de \ce{NH3} (MM = 17~g/mol),
          calcule la masa m\'axima de \ce{NH4NO3}
          (MM = 80~g/mol) que podr\'ia formarse.

    \item Los tres mecanismos de reacci\'on de compuestos
          de carbono son \ce{O(^1D)}, \ce{OH^.} y \ce{O3}.
          Para un alcano (hidrocarburo saturado) y un alqueno
          (insaturado), indique qu\'e mecanismo(s) son
          relevantes en cada caso y explique por qu\'e el
          \ce{O3} no extrae hidr\'ogenos de alcanos.

    \item El PAN (\ce{HC(O)OONO2}) se origina a partir de
          formaldeh\'ido (\ce{HCHO}). Escriba los tres
          pasos de su formaci\'on:
          (i) abstracci\'on del H de \ce{HCHO} por OH,
          (ii) adici\'on de \ce{O2} al radical formilo para
          dar el radical peroxiformilo \ce{HC(O)OO^.},
          (iii) combinaci\'on con \ce{NO2} para producir PAN.

    \item Explique por qu\'e el PAN es especialmente da\~nino
          para las plantas (lacrimos\'ageno) y por qu\'e
          su concentraci\'on es m\'axima durante episodios
          de esmog fotoqu\'imico de mediod\'ia.
\end{enumerate}

\textit{[Respuesta:
(a) moles \ce{NH3} = $10^5/17 = 5882$~mol;
    masa \ce{NH4NO3} = $5882\times80
    \approx \SI{471}{\kilo\gram}$]}

% ── E8: Halógenos — ciclo Cl, reactividad y CFCs ───────────────────────────
\item Los \'atomos de hal\'ogeno siguen dos rutas
competitivas (\ref{RH1} y \ref{RH2}).

\begin{enumerate}[label=\alph*)]
    \item La fracci\'on que reacciona v\'ia \ref{RH2}
          (oxidaci\'on por \ce{O3}) es F~$\sim$0\%,
          Cl~50\%, Br~99\%, I~100\%.
          Explique la tendencia creciente de F a I en
          t\'erminos de la disminuci\'on de la energ\'ia
          de enlace H--X al bajar en el grupo 17 de la
          tabla peri\'odica.

    \item Para la secuencia de liberaci\'on de Cl de \ce{CH3Cl}:
          \ce{CH3Cl + OH^. -> ^.CH2Cl + H2O},
          \ce{^.CH2Cl + O2 ->[M] CH2ClOO^.},
          \ce{CH2ClOO^. + NO -> NO2 + HCHO + Cl^.},
          identifique (i) el paso que regenera Cl libre,
          (ii) el que oxida NO a \ce{NO2},
          (iii) el producto carbonilo formado.

    \item La reacci\'on con aerosoles de sal marina
          \ce{H2SO4 + 2NaCl_{(s)} -> 2HCl + Na2SO4_{(s)}}
          libera HCl gaseoso. Calcule la masa de HCl
          (MM = 36.5~g/mol) producida a partir de
          \SI{1}{\gram} de \ce{H2SO4} (MM = 98~g/mol).

    \item Explique por qu\'e los CFCs son inertes en
          la troposfera pero se fotolisan en la estratosfera,
          y por qu\'e el Cl liberado puede destruir hasta
          $10^5$ mol\'eculas de \ce{O3} antes de ser
          neutralizado como HCl.
\end{enumerate}

\textit{[Respuesta:
(c) moles \ce{H2SO4} = $1/98 = 0.01020$~mol;
    moles HCl = $2\times0.01020 = 0.02041$~mol;
    masa HCl = $0.02041\times36.5
    = \SI{0.745}{\gram}$]}

\end{ejercicios}
