% ── E1: Distribución de ozono y circulación Brewer-Dobson ──────────────────
\item La circulaci\'on de Brewer-Dobson transporta ozono
desde los tr\'opicos hacia los polos.

\begin{enumerate}[label=\alph*)]
    \item La ecuaci\'on de continuidad del ozono es:
          \[
              \frac{\partial [\ce{O3}]}{\partial t}
              = P - L - \nabla\cdot(\vec{v}[\ce{O3}])
          \]
          Indique el significado f\'isico de cada t\'ermino
          ($P$, $L$ y $\nabla\cdot(\vec{v}[\ce{O3}])$)
          y explique por qu\'e $P > 0$ en los tr\'opicos
          mientras que $\nabla\cdot(\vec{v}[\ce{O3}]) < 0$
          en los polos (transporte neto hacia los polos).

    \item El ozono se produce principalmente en la
          estratosfera tropical por fot\'olisis de \ce{O2}
          a $\lambda < \SI{242}{\nano\metre}$,
          pero los m\'aximos de columna se observan a
          latitudes altas ($>60^\circ$).
          Explique esta aparente paradoja en t\'erminos
          de la circulaci\'on de Brewer-Dobson.

    \item El gradiente vertical de ozono se invierte
          entre $\sim\SI{15}{\kilo\metre}$
          y $\sim\SI{30}{\kilo\metre}$.
          Explique qu\'e procesos dominan la destrucci\'on
          en capas bajas de la estratosfera y cu\'al es
          la fuente dominante a \SI{30}{\kilo\metre}.

    \item Las asimetr\'ias hemisf\'ericas hacen que
          el hemisferio norte tenga mayor columna de ozono
          que el sur (fuera del agujero ant\'artico).
          ?`Qu\'e caracter\'istica geogr\'afica del
          hemisferio norte amplifica las ondas
          orogr\'aficas y aumenta el transporte de ozono?
\end{enumerate}

% ── E2: Reacciones de Chapman — balance natural ────────────────────────────
\item Las reacciones de Chapman (\ref{OE1}--\ref{OE3})
describen el equilibrio natural del ozono estratosf\'erico.

\begin{enumerate}[label=\alph*)]
    \item Escriba las cuatro reacciones del ciclo de Chapman:
          (i) fot\'olisis de \ce{O2} ($\lambda<242$~nm),
          (ii) formaci\'on de \ce{O3} (con M),
          (iii) fot\'olisis de \ce{O3} ($\lambda<310$~nm),
          (iv) reacci\'on \ce{O3 + O -> 2O2}.
          Indique cu\'ales producen y cu\'ales destruyen \ce{O3}.

    \item Aplique la PSSA al \'atomo \ce{O^.}:
          iguale su tasa de formaci\'on (reacciones i y iii)
          a su tasa de p\'erdida (reacciones ii y iv)
          para obtener la expresi\'on de $[\ce{O}]_\mathrm{ss}$
          en funci\'on de $[\ce{O2}]$, $[\ce{O3}]$ y las
          constantes de velocidad.

    \item El ciclo de Chapman predice concentraciones de
          ozono mayores que las observadas.
          Nombre los tres tipos de ciclos catal\'iticos
          (secci\'on \ref{cpeoz}) que explican el ``sumidero
          faltante'' e indique la familia radical
          responsable de cada uno.

    \item La reacci\'on \ce{O2 + h{\nu} -> 2O^.} solo ocurre
          a $\lambda < \SI{242}{\nano\metre}$.
          Explique por qu\'e esta radiaci\'on es absorbida
          completamente en la estratosfera y no llega
          a la troposfera.
\end{enumerate}

% ── E3: Ciclo catalítico HOₓ ──────────────────────────────────────────────
\item El ciclo \ce{HO_$x$} es el primer sumidero catal\'itico
identificado para el \ce{O3} estratosf\'erico.

\begin{enumerate}[label=\alph*)]
    \item Escriba el ciclo completo:
          \ce{OH + O3 -> HO2 + O2} y
          \ce{HO2 + O3 -> OH + 2O2},
          y verifique que el neto es \ce{2O3 -> 3O2}.

    \item La terminaci\'on del ciclo requiere la reacci\'on
          \ce{OH + HO2 -> H2O + O2}.
          ?`Por qu\'e esta reacci\'on extingue el ciclo
          en lugar de propagarlo?

    \item El \ce{H2O} estratosf\'erico proviene del
          transporte y de la oxidaci\'on del \ce{CH4}.
          Escriba la reacci\'on de producci\'on de OH
          a partir de \ce{O(^1D)} y \ce{H2O}
          (reacci\'on \ref{OHr1}) y explique por qu\'e
          se necesita \ce{O(^1D)} en lugar de \ce{O(^3P)}.

    \item Si una mol\'ecula de \ce{H2O} inicia el ciclo
          produciendo 2~OH, y cada par OH/\ce{HO2} destruye
          2~mol\'eculas de \ce{O3} por ciclo antes de
          terminarse, ?`cu\'antas mol\'eculas de \ce{O3}
          destruye el ciclo si completa en promedio
          500~vueltas antes de la terminaci\'on?
\end{enumerate}

\textit{[Respuesta:
(d) $2\times500 = 1\,000$~mol\'ec de \ce{O3} por
    mol\'ecula de \ce{H2O} iniciadora]}

% ── E4: Ciclo catalítico NOₓ — aviones supersónicos ───────────────────────
\item El ciclo \ce{NO_$x$} fue identificado al evaluar
las emisiones de aviones supers\'onicos en la estratosfera.

\begin{enumerate}[label=\alph*)]
    \item Escriba el ciclo catal\'itico \ce{NO_x} completo
          (reacciones \ref{R10} y \ref{R11}) e indique
          el neto: \ce{O3 + O^. -> 2O2}.

    \item El ciclo entre NO y \ce{NO2} por las reacciones:
          \ce{NO + O3 -> NO2 + O2} y
          \ce{NO2 + h{\nu} -> NO + O^.},
          \ce{O^. + O2 ->[M] O3}
          se denomina ciclo nulo.
          ?`Por qu\'e no tiene efecto neto sobre el \ce{O3}?
          ?`Cu\'al es su importancia en la determinaci\'on
          de la relaci\'on \ce{NO}/\ce{NO2}?

    \item Durante el d\'ia, el \ce{NO_$x$} se termina por:
          \ce{NO2 + OH ->[M] HNO3}.
          Durante la noche, la v\'ia es:
          \ce{NO2 + O3 -> NO3 + O2},
          \ce{NO3 + NO2 ->[M] N2O5}.
          Explique por qu\'e el \ce{N2O5} solo se acumula
          de noche.

    \item El \ce{N2O} produce NO en la estratosfera v\'ia
          \ce{N2O + O(^1D) -> 2NO}.
          Esta reacci\'on representa el 5\% de la p\'erdida
          estratosf\'erica de \ce{N2O}.
          Si las emisiones globales de \ce{N2O} son
          $\approx18$~Tg~N/a\~no, estime la producci\'on
          anual de NO activo en la estratosfera
          (MM de N = 14~g/mol;
          MM de NO = 30~g/mol).
\end{enumerate}

\textit{[Respuesta:
(d) \ce{N2O} a NO: $0.05\times18 = 0.9$~Tg~N/a\~no;
    en masa de NO: $0.9\times(30/14)
    \approx 1.93$~Tg~NO/a\~no]}

% ── E5: Ciclo catalítico ClOₓ y CFCs ──────────────────────────────────────
\item En 1974 Molina y Rowland identificaron el peligro
de los CFCs para el ozono estratosf\'erico.

\begin{enumerate}[label=\alph*)]
    \item El CFC-12 (\ce{CF2Cl2}, MM = 121~g/mol)
          libera \'atomos de Cl al fotolizarse.
          Si se emiten \SI{1}{\kilo\gram} de CFC-12,
          calcule cu\'antos \'atomos de Cl se liberan
          ($N_A = 6.022\times10^{23}$~mol\'ec/mol)
          y cu\'antas mol\'eculas de \ce{O3} podr\'ian
          destruirse si cada \'atomo activo completa
          $10^5$ ciclos catal\'iticos.

    \item Escriba el ciclo \ce{ClO_$x$} completo:
          \ce{Cl + O3 -> ClO + O2} y
          \ce{ClO + O^. -> Cl + O2},
          y verifique que el neto coincide con el del
          ciclo \ce{NO_$x$}: \ce{O3 + O^. -> 2O2}.

    \item La terminaci\'on del ciclo \ce{ClO_$x$} produce
          los reservorios HCl y \ce{ClNO3}:
          \ce{Cl + CH4 -> HCl + CH3^.} y
          \ce{ClO + NO2 ->[M] ClNO3}.
          Calcule el tiempo de vida del HCl
          con $k_{\ce{HCl+OH}} = 8\times10^{-13}$
          cm$^3$~mol\'ec$^{-1}$~s$^{-1}$ y
          $[\ce{OH}] = 5\times10^5$~mol\'ec/cm$^3$.

    \item Explique por qu\'e Molina, Rowland y Crutzen
          recibieron el premio Nobel de Qu\'imica en 1995,
          destacando la contribuci\'on espec\'ifica de
          cada uno seg\'un el cap\'itulo.
\end{enumerate}

\textit{[Respuesta:
(a) moles CFC-12 $= 1000/121 = 8.26$~mol;
    \'atomos Cl $= 2\times8.26\times6.022\times10^{23}
    = 9.95\times10^{24}$;
    mol\'ec \ce{O3} $= 9.95\times10^{24}\times10^5
    = 9.95\times10^{29}$;
(c) $\tau_\mathrm{HCl} = 1/(8\times10^{-13}
    \times5\times10^5) = 2.5\times10^6$~s $\approx29$~d]}

% ── E6: Ciclos 2 y 3 del agujero antártico ─────────────────────────────────
\item El \textbf{Cuadro~\ref{desozo}} muestra tres ciclos
de destrucci\'on de ozono con hal\'ogenos.

\begin{enumerate}[label=\alph*)]
    \item Complete la tabla de los tres ciclos,
          verificando que todos producen el mismo neto:

{%
\centering
\begin{tabular}{llll}
\toprule
\textbf{Ciclo} & \textbf{Reactivo clave} &
\textbf{Intermediario} & \textbf{Neto} \\
\midrule
1 & \ce{Cl + O3}  & \ce{ClO} & \ce{O3 + O^. -> 2O2} \\
2 & \ce{ClO + ClO} & \ce{(ClO)2} & \underline{\hspace{2cm}} \\
3 & \ce{ClO + BrO} & \ce{Cl + Br} & \underline{\hspace{2cm}} \\
\bottomrule
\end{tabular}
\par
}%

    \item El ciclo~2 involucra la dimerizaci\'on
          \ce{ClO + ClO -> (ClO)2}. Esta reacci\'on
          es importante en el agujero ant\'artico porque
          $[\ce{ClO}]$ es much\'isimo mayor que en
          latitudes medias. Explique por qu\'e las PSC
          (nubes estratosf\'ericas polares) provocan
          ese aumento de \ce{ClO} activo.

    \item El ciclo~3 muestra sinergia entre Cl y Br:
          la reacci\'on \ce{ClO + BrO -> Cl + Br + O2}
          destruye \ce{O3} m\'as eficientemente que
          cada hal\'ogeno solo.
          Si la concentraci\'on de Br es 200 veces menor
          que la de Cl, ?`por qu\'e el Br contribuye
          significativamente a la destrucci\'on total?

    \item Las PSC se forman a $T < \SI{-72}{\celsius}$.
          Explique por qu\'e el hemisferio sur forma
          PSC mucho m\'as frecuentemente que el norte
          y en qu\'e \'epoca del a\~no ocurre el m\'aximo
          agotamiento de \ce{O3} sobre la Ant\'artida.
\end{enumerate}

\textit{[Respuesta: (c) el Br reacciona casi exclusivamente
con \ce{O3} v\'ia \ce{RH2} (99\% frente a s\'olo 50\%
del Cl), haciendo que cada \'atomo de Br sea mucho
m\'as eficiente por ciclo que el Cl]}

% ── E7: PSC y reactivación del cloro ──────────────────────────────────────
\item La qu\'imica de superficie de las PSC convierte
el cloro inactivo en formas reactivas.

\begin{enumerate}[label=\alph*)]
    \item En la superficie de una PSC ocurre:
          \ce{HCl + ClNO3 ->[\text{PSC}] Cl2 + HNO3}.
          Explique por qu\'e esta reacci\'on heterog\'enea
          es mucho m\'as r\'apida que la correspondiente
          en fase gas, invocando la concentraci\'on
          superficial de reactivos.

    \item Cuando la luz solar regresa en primavera
          austral, el \ce{Cl2} se fotoliza instantaneamente:
          \ce{Cl2 + h{\nu} -> 2Cl^.}.
          ?`Por qu\'e este paso desencadena una destrucci\'on
          masiva y r\'apida de \ce{O3}?
          ?`Qu\'e le ocurre al \ce{HNO3} formado en la PSC?

    \item Bajo las condiciones del agujero ant\'artico
          ($T < \SI{-72}{\celsius}$, alta $[\ce{Cl}]$,
          escasa radiaci\'on UV para el ciclo de Chapman),
          analice cu\'al de los tres t\'erminos de la
          ecuaci\'on de continuidad
          ($P$, $L$, $\nabla\cdot\vec{v}[\ce{O3}]$)
          domina y en qu\'e direcci\'on cambia $[\ce{O3}]$.

    \item El Protocolo de Montreal (1987) redujo la
          producci\'on de CFCs. Si la concentraci\'on
          estratosf\'erica de Cl lleg\'o a $\approx3.5$~ppb
          en el a\~no 2000 y decrece al $0.7\%$/a\~no,
          estime en qu\'e a\~no volver\'ia a los niveles
          preindustriales ($\approx0.6$~ppb) y
          comente la coherencia con los tiempos de vida
          de los CFCs del \textbf{Cuadro~\ref{WGP}}
          del cap\'itulo 5.
\end{enumerate}

\textit{[Respuesta:
(d) reducci\'on necesaria: $(3.5-0.6)/3.5 = 82.9\%$;
    a tasa constante del 0.7\%/a\~no:
    $\ln(3.5/0.6)/0.007 \approx 254$~a\~nos desde el
    m\'aximo $\Rightarrow$ $\approx$ a\~no 2254;
    coherente con vidas de CFCs de 45--1700~a\~nos]}

% ── E8: Unidades Dobson — conversiones y agujero antártico ─────────────────
\item La Unidad Dobson (DU) es la medida est\'andar
de la columna total de ozono:
1~DU $\equiv$ $2.69\times10^{16}$~mol\'ec/cm$^2$
$\equiv$ 0.01~mm de ozono puro a \SI{0}{\celsius}
y 1~atm.

\begin{enumerate}[label=\alph*)]
    \item La columna normal es 300~DU.
          Calcule (i) el n\'umero total de mol\'eculas de
          \ce{O3} en una columna de 1~cm$^2$ de base,
          y (ii) el espesor f\'isico equivalente en mm.

    \item Durante el agujero ant\'artico la columna cae
          a $\approx$100~DU. Calcule:
          (i) la reducci\'on porcentual respecto al valor
          normal,
          (ii) el n\'umero de mol\'eculas de \ce{O3}
          perdidas por cm$^2$,
          (iii) la masa de \ce{O3} perdida por cada
          \si{\kilo\metre\squared} de superficie afectada
          (MM \ce{O3} = 48~g/mol).

    \item Todo el aire atmosf\'erico comprimido a
          \SI{0}{\celsius} y 1~atm formar\'ia una columna
          de \SI{8}{\kilo\metre}. Compare este valor con
          los 3~mm de la capa de ozono y exprese la
          fracci\'on de ozono como partes por mill\'on
          en volumen (ppmv).

    \item Por cada 1\% de disminuci\'on de ozono
          la radiaci\'on UV-B aumenta $\approx$2\%.
          ?`Cu\'anto aumenta la UV-B cuando la columna
          pasa de 300 a 100~DU? Comente las consecuencias
          biol\'ogicas mencionadas en el cap\'itulo.
\end{enumerate}

\textit{[Respuesta:
(a-i) $300\times2.69\times10^{16}
      = 8.07\times10^{18}$~mol\'ec/cm$^2$;
(a-ii) $300\times0.01 = 3$~mm;
(b-i) reducci\'on $= (300-100)/300 = 66.7\%$;
(b-ii) $200\times2.69\times10^{16}
       = 5.38\times10^{18}$~mol\'ec/cm$^2$;
(b-iii) moles/km$^2$ $= 5.38\times10^{18}
        \times10^{10}/6.022\times10^{23}
        = 8.93\times10^4$~mol;
    masa $= 8.93\times10^4\times48
        = 4.29\times10^6$~g/km$^2$
        $\approx\SI{4.3}{\tonne\per\kilo\metre\squared}$;
(c) $3\times10^{-3}/(8\times10^3) = 3.75\times10^{-7}
    = 0.375$~ppmv;
(d) aumento UV-B $= 66.7\times2 = 133\%$]}
